export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') return res.status(200).end(); const MOTIVE_KEY = process.env.MOTIVE_API_KEY; const SB_URL = 'https://ghrbrwusfyjprxhivxfo.supabase.co'; const SB_KEY = 'sb_publishable_NAFw4krk8evjh7Bx0VFD9g_EcuThMES'; try { // Try vehicle locations endpoint which has odometer data const motiveRes = await fetch('https://api.gomotive.com/v1/vehicle_locations?per_page=100', { headers: { 'X-Api-Key': MOTIVE_KEY, 'Content-Type': 'application/json' } }); if (!motiveRes.ok) { const err = await motiveRes.text(); // Try vehicles endpoint as fallback const fallbackRes = await fetch('https://api.gomotive.com/v1/vehicles?per_page=100', { headers: { 'X-Api-Key': MOTIVE_KEY, 'Content-Type': 'application/json' } }); const fallbackErr = await fallbackRes.text(); return res.status(500).json({ error: 'Motive API error', vehicle_locations_error: err, vehicles_error: fallbackErr }); } const motiveData = await motiveRes.json(); const vehicles = motiveData.vehicles || motiveData.vehicle_locations || []; const results = []; for (const v of vehicles) { const vehicle = v.vehicle || v; const vehicleId = vehicle.id?.toString() || vehicle.vehicle_id?.toString(); // Try different odometer fields let odometer = null; if (vehicle.current_meter_millimeters) { odometer = Math.round(vehicle.current_meter_millimeters / 1609344); } else if (vehicle.odometer) { odometer = Math.round(vehicle.odometer); } else if (vehicle.meter) { odometer = Math.round(vehicle.meter / 1609.344); } if (!vehicleId) continue; // Sync to Supabase const syncRes = await fetch(`${SB_URL}/rest/v1/rpc/sync_motive_odometer`, { method: 'POST', headers: { 'Content-Type': 'application/json', 'apikey': SB_KEY, 'Authorization': `Bearer ${SB_KEY}` }, body: JSON.stringify({ p_truck_id: vehicleId, p_motive_vehicle_id: vehicleId, p_odometer_miles: odometer || 0 }) }); results.push({ vehicle_id: vehicleId, odometer_miles: odometer, synced: syncRes.ok }); } return res.status(200).json({ success: true, vehicles_found: vehicles.length, synced: results.length, results, raw_sample: vehicles[0] || null }); } catch (err) { return res.status(500).json({ error: err.message }); } }
