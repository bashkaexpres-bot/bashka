RPC: upsert a driver medical/license record create or replace function dqf_import_driver_medical( p_user_id uuid, p_full_name text, p_date_of_birth date, p_license_number text, p_license_state text, p_examiner text, p_cert_state text, p_issue_date date, p_expiry_date date, p_status text ) returns json language plpgsql security definer set search_path = public as $$ declare v_role text; begin -- confirm caller is admin or has dqf access, matching fleet_users select role into v_role from fleet_users where id = p_user_id; if v_role is null then return json_build_object('error', 'Not authorized'); end if; insert into driver_medical_records ( full_name, date_of_birth, license_number, license_state, examiner, cert_state, issue_date, expiry_date, status, updated_at ) values ( p_full_name, p_date_of_birth, p_license_number, p_license_state, p_examiner, p_cert_state, p_issue_date, p_expiry_date, p_status, now() ) on conflict (full_name, license_number) do update set date_of_birth = excluded.date_of_birth, license_state = excluded.license_state, examiner = excluded.examiner, cert_state = excluded.cert_state, issue_date = excluded.issue_date, expiry_date = excluded.expiry_date, status = excluded.status, updated_at = now(); return json_build_object('success', true); end; $$; -- 4. RPC: upsert a truck's ELD/vehicle info create or replace function dqf_import_truck_eld( p_user_id uuid, p_truck_id text, p_vin text, p_plate text, p_eld_manufacturer text, p_eld_device_name text, p_eld_model_number text, p_eld_software_version text, p_eld_serial_number text ) returns json language plpgsql security definer set search_path = public as $$ declare v_role text; begin select role into v_role from fleet_users where id = p_user_id; if v_role is null then return json_build_object('error', 'Not authorized'); end if; insert into truck_eld_info ( truck_id, vin, plate, eld_manufacturer, eld_device_name, eld_model_number, eld_software_version, eld_serial_number, updated_at ) values ( p_truck_id, p_vin, p_plate, p_eld_manufacturer, p_eld_device_name, p_eld_model_number, p_eld_software_version, p_eld_serial_number, now() ) on conflict (truck_id) do update set vin = excluded.vin, plate = excluded.plate, eld_manufacturer = excluded.eld_manufacturer, eld_device_name = excluded.eld_device_name, eld_model_number = excluded.eld_model_number, eld_software_version = excluded.eld_software_version, eld_serial_number = excluded.eld_serial_number, updated_at = now(); return json_build_object('success', true); end; $$; -- ============================================================ -- NOTE: If you already have tables for drivers or truck/vehicle -- info (e.g. from the earlier compliance import you built), you -- may want to point these RPCs at those existing tables instead -- of creating new ones — just swap the table names above to -- match your schema so you don't end up with duplicate data. -- ============================================================
